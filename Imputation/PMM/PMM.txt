nearest_valid_missing_composition <- function(raw, target, step = 5) {
  
  raw <- as.numeric(raw)
  target <- as.numeric(target)
  
  if (any(is.na(raw))) {
    stop("MICE returned NA for one or more composition values.")
  }
  
  if (is.na(target)) {
    stop("Cannot repair composition because the target sum is NA.")
  }
  
  if (target < -1e-8) {
    stop(
      "Observed AMR/BR/CR cells in at least one partially missing row ",
      "already sum to more than 100. Cannot repair without changing observed data."
    )
  }
  
  target <- max(target, 0)
  k <- length(raw)
  

  if (abs(target / step - round(target / step)) < 1e-8) {
    
    target_units <- round(target / step)
    
    if (k == 1) {
      return(c(target))
    }
    
    if (k == 2) {
      a <- 0:target_units
      candidates <- cbind(a, target_units - a) * step
    } else if (k == 3) {
      candidates <- do.call(
        rbind,
        lapply(0:target_units, function(a) {
          b <- 0:(target_units - a)
          cbind(a, b, target_units - a - b)
        })
      ) * step
    } else {
    
      rounded <- pmax(round(raw / step) * step, 0)
      diff <- target - sum(rounded)
      rounded[which.max(rounded)] <- rounded[which.max(rounded)] + diff
      return(rounded)
    }
    
    raw_matrix <- matrix(
      raw,
      nrow = nrow(candidates),
      ncol = length(raw),
      byrow = TRUE
    )
    
    best_index <- which.min(rowSums((candidates - raw_matrix)^2))
    best <- candidates[best_index, ]
    
    return(as.numeric(best))
  }
  

  rounded <- pmax(round(raw / step) * step, 0)
  diff <- target - sum(rounded)
  adjust_idx <- which.max(rounded)
  rounded[adjust_idx] <- rounded[adjust_idx] + diff
  
  if (any(rounded < -1e-8)) {
    positive_raw <- pmax(raw, 0)
    
    if (sum(positive_raw) > 0) {
      rounded <- positive_raw / sum(positive_raw) * target
    } else {
      rounded <- rep(target / length(raw), length(raw))
    }
  }
  
  return(rounded)
}

adjust_composition_cellwise <- function(
  amr, br, cr,
  miss_amr, miss_br, miss_cr
) {
  
  vals <- c(AMR = amr, BR = br, CR = cr)
  miss <- c(AMR = miss_amr, BR = miss_br, CR = miss_cr)
  
    if (!any(miss)) {
    return(vals)
  }
  

  if (any(is.na(vals[!miss]))) {
    stop("An observed composition cell is unexpectedly NA.")
  }
  
  observed_sum <- sum(vals[!miss])
  missing_target <- 100 - observed_sum
  
  vals[miss] <- nearest_valid_missing_composition(
    raw = vals[miss],
    target = missing_target,
    step = 5
  )

  vals[abs(vals) < 1e-8] <- 0
  
  return(vals)
}

observed_cells_unchanged <- function(after, before, observed_mask, tolerance = 1e-8) {
  all(abs(as.numeric(after[observed_mask]) - as.numeric(before[observed_mask])) <= tolerance)
}

  stopifnot(observed_cells_unchanged(imputed_i$BR,  data$BR,  !miss_BR))
